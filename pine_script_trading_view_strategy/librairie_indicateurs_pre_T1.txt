export Ecart_Bollinger_borne(float Prix = close, int longueurBB = 20,  float upper_band, float lower_band, int longueur_mediane = 100, float coef_mediane = 1, int Nb_bars_above = 6) =>
    // [upperBand, lowerBand, basis] = ta.bb(close, longueurBB, StDev)
    ecart = upper_band - lower_band
    ecart_borne1 = ecart / ta.sma(Prix, longueurBB)
    moyenne_ecart = ta.sma(ecart_borne1, 7)
    mediane = ta.median(ecart_borne1, longueur_mediane) / coef_mediane
    Nb_bars_above_signal = ta.barssince(ecart_borne1 < mediane) >= Nb_bars_above ? 1 : 0
    [ecart_borne1, moyenne_ecart, mediane, Nb_bars_above_signal]


export cross_bbw_low_signal(float Prix = close, int longueurBB = 20, simple float StDev = 2, int fenetre_lowest = 120, float seuil_lowest = 1.4) =>
    LargeurBB = ta.bbw(Prix, longueurBB, StDev)
    bbw_lowest = ta.lowest(LargeurBB, fenetre_lowest)
    cross_bbw_low_signal = LargeurBB <= bbw_lowest * seuil_lowest ? 1 : na
    [LargeurBB, bbw_lowest, cross_bbw_low_signal]

export NB_bars_under_BBW(int NB_bars_under_BBW_mini = 3) =>
    NB_bars_under_BBW = 0
    [LargeurBB, bbw_lowest, cross_bbw_low_signal] = cross_bbw_low_signal()
    for i = 1 to NB_bars_under_BBW_mini by 1
        if bool(cross_bbw_low_signal) and bool(cross_bbw_low_signal[i])
            NB_bars_under_BBW := NB_bars_under_BBW + 1
            NB_bars_under_BBW
    if NB_bars_under_BBW >= NB_bars_under_BBW_mini
        NB_bars_under_BBW_signal = 1
        NB_bars_under_BBW_signal

export bollingerHorizontal(int duree_honriz = 1, int longueurBB = 20, simple float StDev = 2, float Prix = close) =>
    [middle, upper, lower] = ta.bb(Prix, longueurBB, StDev)
    UT_majeur = timeframe.period
    int duree_honriz_corrigee = 0
    if UT_majeur == 'W'
        duree_honriz_corrigee := 10080
        duree_honriz_corrigee
    if UT_majeur == 'D'
        duree_honriz_corrigee := 1440
        duree_honriz_corrigee
    if UT_majeur == '240'
        duree_honriz_corrigee := 240
        duree_honriz_corrigee
    if UT_majeur == '60'
        duree_honriz_corrigee := 60
        duree_honriz_corrigee
    if UT_majeur == '30'
        duree_honriz_corrigee := 30
        duree_honriz_corrigee
    if UT_majeur == '10'
        duree_honriz_corrigee := 10
        duree_honriz_corrigee
    if UT_majeur == '5'
        duree_honriz_corrigee := 5
        duree_honriz_corrigee

    coeffupper = 20000 / upper
    coefflower = 20000 / lower

    float slopeUpper = math.todegrees(math.atan(ta.change(upper, duree_honriz) * coeffupper / duree_honriz_corrigee))
    float slopeLower = math.todegrees(math.atan(ta.change(lower, duree_honriz) * coefflower / duree_honriz_corrigee))
    float slopeMiddle = math.todegrees(math.atan(ta.change(middle, duree_honriz) * coefflower / duree_honriz_corrigee))
    // bollingerHorizontal = (ta.cross(slopeUpper, 0) and ta.cross(slopeLower, 0)) or  (ta.cross(slopeUpper, 0) and ta.cross(slopeLower, 0)[1]) or  (ta.cross(slopeUpper, 0)[1] and ta.cross(slopeLower, 0))? 1 : na
    bollingerHorizontal = ta.cross(slopeUpper, 0) and ta.cross(slopeLower, 0) ? 1 : 0 //and ta.cross(slopeMiddle, 0) 
    [slopeUpper, slopeLower, slopeMiddle, bollingerHorizontal]

export historical_volatility(int length = 500) =>
    annual = 365
    per = timeframe.isintraday or timeframe.isdaily and timeframe.multiplier == 1 ? 1 : 7
    hv = 100 * ta.stdev(math.log(close / close[1]), length) * math.sqrt(annual / per)
    [hv]

export bollingerHorizontalV2(float coeff_MM = 2, int duree_honriz = 1, int longueurBB = 20, simple float StDev = 2, float Prix = close) =>
    [middle, upper, lower] = ta.bb(Prix, longueurBB, StDev)
    [hv] = historical_volatility()
    UT_majeur = timeframe.period
    float duree_honriz_corrigee = 0
    if UT_majeur == 'W'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == 'D'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '720'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '360'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '240'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '120'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '60'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '30'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '15'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '10'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '5'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    if UT_majeur == '1'
        duree_honriz_corrigee := 1
        duree_honriz_corrigee
    // duree_honriz_corrigee := duree_honriz_corrigee * 
    coeffupper = 1 //20000 / upper
    coefflower = 1 //20000 / lower
    float slopeUpper = math.todegrees(math.atan(ta.change(upper, 1) * coeffupper / duree_honriz_corrigee))
    float slopeLower = math.todegrees(math.atan(ta.change(lower, 1) * coefflower / duree_honriz_corrigee))
    float slopeMiddle = math.todegrees(math.atan(ta.change(middle, 1) * coefflower / duree_honriz_corrigee))
    MM_slope_BB = ta.sma(math.abs(slopeUpper), 200) / coeff_MM
    MM_slope_Middle = ta.sma(math.abs(slopeMiddle), 200) / coeff_MM

    horiz_BBup = slopeUpper <= MM_slope_BB and slopeUpper >= -MM_slope_BB or ta.crossunder(slopeUpper, MM_slope_BB) or ta.crossover(slopeUpper, -MM_slope_BB) ? 1 : 0
    horiz_BBLow = slopeLower <= MM_slope_BB and slopeLower >= -MM_slope_BB or ta.crossunder(slopeLower, MM_slope_BB) or ta.crossover(slopeLower, -MM_slope_BB) ? 1 : 0
    horiz_BBMiddle = slopeMiddle <= MM_slope_Middle and slopeMiddle >= -MM_slope_Middle or ta.crossunder(slopeMiddle, MM_slope_Middle) or ta.crossover(slopeMiddle, -MM_slope_Middle) ? 1 : 0

    int compteur_BB = 0
    int bollingerHorizontal = 0
    for i = 1 to duree_honriz by 1
        if bool(horiz_BBup) and bool(horiz_BBLow) and bool(horiz_BBup[i - 1]) and bool(horiz_BBLow[i - 1])
            compteur_BB := compteur_BB + 1
            compteur_BB
    if compteur_BB >= duree_honriz
        bollingerHorizontal := 1
        bollingerHorizontal

    int compteur_middle = 0
    int MM20Horizontale = 0
    for j = 1 to duree_honriz by 1
        if bool(horiz_BBMiddle) and bool(horiz_BBMiddle[j - 1])
            compteur_middle := compteur_middle + 1
            compteur_middle
    if compteur_middle >= duree_honriz
        MM20Horizontale := 1
        MM20Horizontale

    [slopeUpper, slopeLower, slopeMiddle, MM_slope_BB, MM_slope_Middle, horiz_BBup, horiz_BBLow, horiz_BBMiddle, bollingerHorizontal, MM20Horizontale]


export bollingerHorizontalV3(float coeff_medianeBBW = 1.1, simple float StDev = 2, int longueurBB = 20, float Prix = close) =>
    [middle, upper, lower] = ta.bb(Prix, longueurBB, StDev)
    BBW = ta.bbw(Prix, longueurBB, StDev)
    MMBBW = ta.sma(BBW, 5)
    medianeBBW = ta.median(BBW, 200)
    seuilBBW = BBW < medianeBBW / coeff_medianeBBW ? 1 : 0
    seuilMMBBW = MMBBW < medianeBBW / coeff_medianeBBW ? 1 : 0
    bollingerHorizontal = bool(seuilBBW) or bool(seuilMMBBW) ? 1 : 0
    [BBW, MMBBW, medianeBBW, seuilBBW, seuilMMBBW, bollingerHorizontal]

export croisement_sma7_sma23_anticip_bull_type_A(int length7 = 7, int length23 = 23, float cours = close) =>
    // croisement des sma 7 et sma 23 avec anticipation (retrait de la 7eme et 23eme bar, ajout hl2)
    var sma7 = array.new<float>(length7, close)
    var sma23 = array.new<float>(length23, close)

    if array.size(sma7) == length7 //and barstate.isconfirmed
        array.remove(sma7, 0)
        array.push(sma7, close)

    if array.size(sma23) == length23 //and barstate.isconfirmed
        array.remove(sma23, 0)
        array.push(sma23, close)

    // Affichage des croisements stochastique avec sa moyenne
    int cross_over_sma_type_A = 0
    float sma7_anticip = (array.sum(sma7) - close[length7] + cours) / length7
    float sma23_anticip = (array.sum(sma23) - close[length23] + cours) / length23

    if barstate.isconfirmed
        cross_over_sma_type_A := sma7_anticip > sma23_anticip and sma7_anticip[1] < sma23_anticip[1] and sma23_anticip > sma23_anticip[1] ? 1 : na
        cross_over_sma_type_A

export Parallele_BB_BBW(float Prix = close, int longueur_change = 1, int coef_seuil = 50, simple float StDev = 2, int longueurBB = 20) =>
    [middle, upper, lower] = ta.bb(Prix, longueurBB, StDev)
    change_upper = math.abs(ta.change(upper, longueur_change) / upper[1] * 100)
    change_lower = -math.abs(ta.change(lower, longueur_change) / lower[1] * 100)
    BBW = ta.bbw(Prix, longueurBB, StDev)
    highest_div = ta.highest(change_upper, 500)
    lowest_div = ta.lowest(change_lower, 500)
    seuil_haut = highest_div / coef_seuil
    seuil_bas = lowest_div / coef_seuil
    if barstate.isconfirmed
        Parallele_BB_BBW = change_upper < seuil_haut and change_lower > seuil_bas and BBW < seuil_haut / 1 ? 1 : 0
        Parallele_BB_BBW

    Position_close_short
    // Position_close_short = close < open and close <= low + ((high - low) / Distance_close)? 1 : 0

export depassement_RoC_long(float Depassement_sma_RoC = 1.5, float RoC_Max_T1 = 3) =>

    depassement_RoC_long

    depassement_RoC_short

export SAR_pente_long() =>
    SAR = ta.sar(0.02, 0.02, 0.2)
    SAR_pente_long = SAR > SAR[1] ? 1 : 0
    SAR_pente_long

export SAR_pente_short() =>
    SAR = ta.sar(0.02, 0.02, 0.2)
    SAR_pente_long = SAR < SAR[1] ? 1 : 0
    SAR_pente_long

export Cross_SAR_SMA20_long(float Prix = close, int longueurBB = 20) =>
    SAR = ta.sar(0.02, 0.02, 0.2)
    SMA20 = ta.sma(Prix, longueurBB)
    Cross_SAR_SMA20_long = ta.crossover(SAR, SMA20)
    Cross_SAR_SMA20_long

export Cross_SAR_SMA20_short(float Prix = close, int longueurBB = 20) =>
    SAR = ta.sar(0.02, 0.02, 0.2)
    SMA20 = ta.sma(Prix, longueurBB)
    Cross_SAR_SMA20_short = ta.crossunder(SAR, SMA20)
    Cross_SAR_SMA20_short
